/* ===== Fullscreen helper (immersive mode) ===== - Click FULLSCREEN or press F - ESC exits (browser standard) - We update button label + body class */ (function(){ const btn = document.getElementById("btnFullscreen"); if(!btn) return; function isFS(){ return !!document.fullscreenElement; } async function toggleFS(){ try{ if(!isFS()){ await document.documentElement.requestFullscreen({ navigationUI: "hide" }); }else{ await document.exitFullscreen(); } }catch(e){ // If fullscreen is blocked, do nothing quietly (classroom machines vary) console.warn("Fullscreen unavailable:", e); } } function syncUI(){ const on = isFS(); document.body.classList.toggle("isFullscreen", on); btn.textContent = on ? "EXIT FULLSCREEN" : "FULLSCREEN"; } btn.addEventListener("click", toggleFS); document.addEventListener("fullscreenchange", syncUI); document.addEventListener("keydown", (e) => { // Don't steal keys while typing const tag = (e.target && e.target.tagName) ? e.target.tagName.toLowerCase() : ""; const typing = (tag === "input" || tag === "textarea" || e.target?.isContentEditable); if(typing) return; if(e.key === "f" || e.key === "F"){ e.preventDefault(); toggleFS(); } }); // initial syncUI(); })();
