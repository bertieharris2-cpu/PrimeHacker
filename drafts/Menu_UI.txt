  <script>
    const status = document.getElementById("status");
    const footerStatus = document.getElementById("footerStatus");
    const objective = document.getElementById("objective");
    const notes = document.getElementById("notes");

    const anomCount = document.getElementById("anomCount");
    const lastPing = document.getElementById("lastPing");
    const signal = document.getElementById("signal");
    const scanner = document.getElementById("scanner");

    const marqueeTextA = document.getElementById("marqueeTextA");
    const marqueeTextB = document.getElementById("marqueeTextB");

    const tickerA = document.getElementById("tickerA");
    const tickerB = document.getElementById("tickerB");

    const btnTheme = document.getElementById("btnTheme");
    const btnAbout = document.getElementById("btnAbout");
    const btnFocus = document.getElementById("btnFocus");

    let ambientOn = true;
    let focusMode = false;

    // --- Marquee content (scan_screen vibe) ---
    // Keep it shortish but repeatable; we duplicate A/B for a seamless loop.
    const telemetryPhrases = [
      "PACKET TRACE: OK",
      "PRIME SIGNAL LOCKED",
      "ENERGY SPIKE: NONE",
      "FIREWALL: STANDBY",
      "NODE MAP: SYNCED",
      "AUTH: PENDING",
      "DATA LOGS: STREAMING",
      "ANOMALY SWEEP: ACTIVE",
      "PORTS: MONITORED",
      "UPLINK: ENCRYPTED"
    ];

    function buildMarqueeLine(){
      // Build one long line with separators so it feels “pro”
      const sep = "   ░▒▓   ";
      return telemetryPhrases.join(sep) + sep + "PRIMENET SCAN LOOP";
    }

    const marqueeLine = buildMarqueeLine();
    marqueeTextA.textContent = marqueeLine;
    marqueeTextB.textContent = marqueeLine;

    // --- Bottom ticker ---
    const tickerItems = [
      "SYSLOG: Boot sequence complete",
      "INFO: Operator console loaded",
      "INFO: Telemetry stream initialised",
      "WARN: No modules linked (expected)",
      "OK: UI integrity nominal"
    ];
    const tickerSep = "   •   ";
    const tickerLine = tickerItems.join(tickerSep) + tickerSep + "PRIME NET READY";
    tickerA.textContent = tickerLine;
    tickerB.textContent = tickerLine;

    // --- Fake “live” stats so it feels alive ---
    function nowTime(){
      const d = new Date();
      const hh = String(d.getHours()).padStart(2,"0");
      const mm = String(d.getMinutes()).padStart(2,"0");
      const ss = String(d.getSeconds()).padStart(2,"0");
      return `${hh}:${mm}:${ss}`;
    }

    function ping(){
      lastPing.textContent = nowTime();
      // tiny chance of anomaly just to show life
      const hit = Math.random() < 0.08;
      if(hit){
        const current = Number(anomCount.textContent || "0");
        anomCount.textContent = String(current + 1);
        signal.textContent = "FLUCTUATING";
        footerStatus.textContent = "ANOMALY FLAGGED";
        setTimeout(() => {
          signal.textContent = "STABLE";
          footerStatus.textContent = "READY";
        }, 900);
      }else{
        footerStatus.textContent = focusMode ? "FOCUS" : "READY";
      }
    }
    setInterval(ping, 1200);
    ping();

    // --- Menu interactions (placeholders only) ---
    const actions = {
      start(){
        status.textContent = "STATUS: BOOTING";
        objective.textContent = "Initiate training handshake.";
        notes.textContent = "Placeholder only. Later, this can fade into your hub / map screen.";
      },
      continue(){
        status.textContent = "STATUS: CHECKPOINT";
        objective.textContent = "Locate last session data.";
        notes.textContent = "No save system yet. We can stub one with localStorage when you're ready.";
      },
      modules(){
        status.textContent = "STATUS: MODULES";
        objective.textContent = "Select a module (locked).";
        notes.textContent = "Next step: add your triangulation, geometry, coordinates, logic gates, data logs, etc.";
      },
      settings(){
        status.textContent = "STATUS: SETTINGS";
        objective.textContent = "Adjust operator preferences.";
        notes.textContent = "We can add toggles for motion reduction, font size, and audio cues.";
      },
      credits(){
        status.textContent = "STATUS: CREDITS";
        objective.textContent = "System authorship and references.";
        notes.textContent = "Placeholder panel. Could become a full-screen overlay like your finale UI.";
      },
      quit(){
        status.textContent = "STATUS: STANDBY";
        objective.textContent = "Session ended (placeholder).";
        notes.textContent = "In a browser you can’t close tabs reliably; we can show a ‘safe to close’ screen instead.";
      }
    };

    document.querySelectorAll(".bigBtn").forEach(btn => {
      btn.addEventListener("click", () => {
        const key = btn.getAttribute("data-action");
        (actions[key] || (()=>{}))();
      });
    });

    // --- Controls ---
    btnTheme.addEventListener("click", () => {
      ambientOn = !ambientOn;
      document.querySelector(".backdrop").style.display = ambientOn ? "block" : "none";
      footerStatus.textContent = ambientOn ? "AMBIENT ON" : "AMBIENT OFF";
      setTimeout(() => footerStatus.textContent = focusMode ? "FOCUS" : "READY", 700);
    });

    btnAbout.addEventListener("click", () => {
      status.textContent = "STATUS: ABOUT";
      objective.textContent = "PRIME NET UI SHELL";
      notes.textContent = "Main menu scaffold. Next: add map screen, then connect minigames one by one.";
    });

    btnFocus.addEventListener("click", () => {
      focusMode = !focusMode;
      document.documentElement.style.setProperty("--fontSize", focusMode ? "20px" : "18px");
      document.documentElement.style.setProperty("--lineHeight", focusMode ? "1.7" : "1.6");
      footerStatus.textContent = focusMode ? "FOCUS" : "READY";
      document.getElementById("opMode").textContent = focusMode ? "MODE: FOCUS" : "MODE: TRAINING";
    });

    // Keyboard niceties (optional)
    document.addEventListener("keydown", (e) => {
      const k = e.key.toLowerCase();
      if(k === "1") actions.start();
      if(k === "2") actions.continue();
      if(k === "3") actions.modules();
      if(k === "4") actions.settings();
      if(k === "f") btnFocus.click();
    });
  </script>
